### home-setup
# Not yet tested. ---- On the admin laptop: a Pi image built from 2026-11-22 (nest, tree) ----
./build.sh nest                          # or tree: the image carries sparrows-home and the LUKS tools
./build.sh sd                            # a card of 4 GB or more: /home starts at 2048 MiB

# ---- First boot made an empty partition for /home. On the unit ----
sparrows-home status                     # "partition /dev/mmcblk0p3 is there, not set up yet"

# ---- Once per unit, from the admin laptop: its passphrase, shown once, then sent over SSH ----
./build.sh home <unit address>           # Enter = a random passphrase; type "saved" once it is saved
ls secrets/home/                         # <unit name>.key: back it up offline with secrets/

# Verify on the unit
sparrows-home status                     # /home is OPEN, with /home/lxmd and /home/nomadnetwork
blkid /dev/mmcblk0p3; df -h /home          # LABEL="sparrows-home" TYPE="crypto_LUKS"; mounted on /home
ls /etc/lxmd /etc/nomadnetwork/storage 2>&1  # gone: the data lives on /home now
/etc/init.d/lxmd status; /etc/init.d/nomadnet status   # running

# After a reboot: locked, the relay works, the message store and the page wait
reboot
sparrows-home status                     # LOCKED
logread | grep 'waits for /home'         # lxmd and nomadnet wait
./build.sh unlock <unit address>         # from the admin laptop, or on the unit: sparrows-unlock

# Before an update: would the image reach /home? (build.sh checks its own images too)
sparrows-home check-image /tmp/fw.bin

# Lock it by hand (stops the message store and the page first)
sparrows-home close

# Undo, back to /etc (the data on /home is lost: copy it out first while it is open)
sparrows-home close
dd if=/dev/zero of=/dev/mmcblk0p3 bs=1M count=16   # wipes the lock and its label: /home is gone for good
reboot                                   # lxmd and nomadnet start fresh in /etc
